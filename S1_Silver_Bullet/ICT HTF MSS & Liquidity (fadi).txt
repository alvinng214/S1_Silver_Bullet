method Validtimeframe(Helper helper, tf) =>
    helper.name := tf
    n1 = timeframe.in_seconds()
    n2 = timeframe.in_seconds(tf)

    n1 < n2

method LineStyle(Helper helper, string style) =>
    helper.name := style

    out = switch style
        '----' => line.style_dashed
        '····' => line.style_dotted
        => line.style_solid

    out

// SkipEQHigh returns the first bar_index of before the equal highs
// This is used to compare pivot points, if two at equal levels
// then we need to compare it to previous high, otherwise it will
// not identify the right STH/STL
method SkipEQHigh(Helper helper, int idx) =>
    helper.name := 'Skip EQ Highs'
    i = idx
    while highs.get(i) == highs.get(i - 1) and i < 5
        i := i + 1
        i
    i
    // SkipEQLows returns the first bar_index of before the equal lows
method SkipEQLow(Helper helper, int idx) =>
    helper.name := 'Skip EQ Lows'
    i = idx
    while lows.get(i) == lows.get(i - 1) and i < 5
        i := i + 1
        i
    i

// Same as above, but works on pivot points. it also doesn't care for
// high or low since Pivot is a single price point
method SkipEQPivot(array<Pivot> p, int idx) =>
    i = idx
    while p.get(i).price == p.get(i - 1).price and p.get(i).lbl_text == p.get(i - 1).lbl_text and p.size() < i
        i := i + 1
        i
    i

method CheckSetup(Setup setup) =>
    if not setup.active and setup.index > 0
        if not setup.hasBOS //TODO we need to account for equal highs and lows when checking for BOS/pivot
            for i = bar_index - setup.index to bar_index - setup.index + 10 by 1 //make 10 configurable 
                if setup.isHigh and low[i + 1] < low[i] and low[i + 1] < low[i + 2] and low[i + 1] < setup.price
                    setup.hasBOS := true
                    setup.BOS_level := low[i + 1]
                    setup.BOS_index := bar_index[i + 1]
                    break
                if setup.isLow and high[i + 1] > high[i] and high[i + 1] > high[i + 2] and high[i + 1] > setup.price
                    setup.hasBOS := true
                    setup.BOS_level := high[i + 1]
                    setup.BOS_index := bar_index[i + 1]
                    setup.BOS_index
        if not setup.hasFVG
            if setup.isHigh and high < low[2] //and low[2] < setup.price
                setup.hasFVG := true
                setup.FVG_open := low[2]
                setup.FVG_close := high
                setup.FVG_index := bar_index[2]
                setup.FVG_index
            if setup.isLow and low > high[2] //and high[2] > setup.price
                setup.hasFVG := true
                setup.FVG_open := high[2]
                setup.FVG_close := low
                setup.FVG_index := bar_index[2]
                setup.FVG_index

        setup.active := setup.hasBOS and setup.hasFVG and setup.isHigh ? close < setup.BOS_level : close > setup.BOS_level and bar_index - setup.pivot.claimed_index <= 10
        if setup.active
            //log.info("Setup Found {0}", setup.isHigh)
            setup.BOS_line := line.new(setup.BOS_index, setup.BOS_level, bar_index, setup.BOS_level)
            setup.FVG_box := box.new(setup.FVG_index, setup.FVG_open, bar_index, setup.FVG_close)
            setup.FVG_box

    // we need rules
    // FVG below the liquidity line
    // BOS below as well
    // 
    setup

// AddLiquidity adds the liquidity lines
method AddLiquidity(MarketStructure MS, Pivot pivot) =>
    if MS.settings.level == 'Short Term' and str.startswith(pivot.lbl_text, 'ST') or MS.settings.level == 'Intermediate Term' and str.startswith(pivot.lbl_text, 'IT') or MS.settings.level == 'Long Term' and str.startswith(pivot.lbl_text, 'LT')
        if na(pivot.ln)
            pivot.ln := line.new(pivot.time, pivot.price, time + (time - time[1]) * MS.settings.extend, pivot.price, xloc = xloc.bar_time, style = helper.LineStyle(MS.settings.liquidity_open_style), color = MS.settings.liquidity_open_color, width = MS.settings.liquidity_open_size)
            lines.unshift(pivot.ln)
            if lines.size() > MS.settings.max_lines
                line l = lines.pop()
                line.delete(l)
        else
            line.set_color(pivot.ln, pivot.claimed ? MS.settings.liquidity_claimed_color : MS.settings.liquidity_open_color)
            line.set_width(pivot.ln, pivot.claimed ? MS.settings.liquidity_claimed_size : MS.settings.liquidity_open_size)
            line.set_x2(pivot.ln, pivot.claimed ? pivot.time_last : time + (time - time[1]) * MS.settings.extend)
            line.set_style(pivot.ln, helper.LineStyle(pivot.claimed ? MS.settings.liquidity_claimed_style : MS.settings.liquidity_open_style))
    MS

// FindIT rechecks the Pivot points and renames the Pivot point from ST to IT
method FindIT(MarketStructure MS) =>
    if MS.STH.size() > 2
        h1 = MS.STH.first()
        h2 = MS.STH.get(1)
        h3 = MS.STH.get(MS.STH.SkipEQPivot(2))

        if h2.price > h3.price and h2.price > h1.price and h2.lbl_text == 'STH'

            h2.lbl_text := 'ITH'
            MS.ITH.unshift(h2)
            MS.AddLiquidity(h2)

    if MS.STL.size() > 2
        l1 = MS.STL.first()
        l2 = MS.STL.get(1)
        l3 = MS.STL.get(MS.STL.SkipEQPivot(2))

        if l2.price < l3.price and l2.price < l1.price and l2.lbl_text == 'STL'
            l2.lbl_text := 'ITL'
            MS.ITL.unshift(l2)
            MS.AddLiquidity(l2)

    MS

// FindLT rechecks the IT Pivot points and renames the Pivot point from IT to LT
method FindLT(MarketStructure MS) =>
    if MS.ITH.size() > 2

        h1 = MS.ITH.first()
        h2 = MS.ITH.get(1)
        h3 = MS.ITH.get(MS.ITH.SkipEQPivot(2))

        if h2.price > h3.price and h2.price > h1.price and h2.lbl_text == 'ITH'
            h2.lbl_text := 'LTH'
            MS.LT.unshift(h2)
            MS.AddLiquidity(h2)

    if MS.ITL.size() > 2
        l1 = MS.ITL.first()
        l2 = MS.ITL.get(1)
        l3 = MS.ITL.get(MS.ITL.SkipEQPivot(2))

        if l2.price < l3.price and l2.price < l1.price and l2.lbl_text == 'ITL'
            l2.lbl_text := 'LTL'
            MS.LT.unshift(l2)
            MS.AddLiquidity(l2)
    MS

// Add method handles the addition of newly discovered ST[H/L] Pivot Point
method Add(MarketStructure MS, float p_price, int p_time, bool isHigh, bool isLow, string lbl) =>
    bool isNew = true

    if MS.ST.size() > 0
        Pivot previous = MS.ST.first()
        isNew := previous.time < p_time
        isNew

    if isNew
        Pivot pivot = Pivot.new()
        pivot.price := p_price
        pivot.time := p_time
        pivot.isHigh := isHigh
        pivot.isLow := isLow
        pivot.lbl_text := lbl


        MS.ST.unshift(pivot)
        if isHigh
            MS.STH.unshift(pivot)
        else
            MS.STL.unshift(pivot)

        MS.AddLiquidity(pivot).FindIT().FindLT()

        if MS.ST.size() > MAX_BUFFER
            Pivot temp = MS.ST.pop()
            line.delete(temp.ln)
    MS


method findST(MarketStructure MS, array<float> h, array<float> l, array<int> t) =>
    _h = h.get(1) > h.get(helper.SkipEQHigh(2)) and h.get(1) > h.first()
    _l = l.get(1) < l.get(helper.SkipEQLow(2)) and l.get(1) < l.first()

    if _h
        MS.Add(h.get(1), t.get(1), true, false, 'STH')
    if _l
        MS.Add(l.get(1), t.get(1), false, true, 'STL')

    MS

// CheckClaimed checks if liquidity has been swept
method CheckClaimed(MarketStructure MS) =>
    if MS.ST.size() > 0
        for i = 0 to MS.ST.size() - 1 by 1
            pivot = MS.ST.get(i)
            if not na(pivot.ln)
                if not pivot.claimed
                    if pivot.isHigh
                        pivot.claimed := close > pivot.price
                        pivot.claimed
                    if pivot.isLow
                        pivot.claimed := close < pivot.price
                        pivot.claimed

                    if pivot.claimed
                        pivot.time_last := time
                        pivot.claimed_index := bar_index
                        pivot.claimed := true

                        line.set_color(pivot.ln, pivot.claimed ? MS.settings.liquidity_claimed_color : MS.settings.liquidity_open_color)
                        line.set_width(pivot.ln, pivot.claimed ? MS.settings.liquidity_claimed_size : MS.settings.liquidity_open_size)
                        line.set_x2(pivot.ln, pivot.claimed ? pivot.time_last : time + (time - time[1]) * MS.settings.extend)
                        line.set_style(pivot.ln, helper.LineStyle(pivot.claimed ? MS.settings.liquidity_claimed_style : MS.settings.liquidity_open_style))

                    else
                        line.set_x2(pivot.ln, time + (time - time[1]) * MS.settings.extend)
                if pivot.claimed and not pivot.reclaimed
                    if pivot.isHigh
                        pivot.reclaimed := close < pivot.price
                        pivot.reclaimed
                    if pivot.isLow
                        pivot.reclaimed := close > pivot.price
                        pivot.reclaimed

                    // TODO; we need to invalidate all previous claimed with lower high or higher low since they are old
                    if pivot.reclaimed
                        line.set_color(pivot.ln, color.red)
                        setup.time := time
                        setup.index := bar_index
                        setup.price := pivot.price
                        setup.hasBOS := false
                        setup.hasFVG := false
                        setup.FVG_open := 0
                        setup.FVG_close := 0
                        setup.isHigh := pivot.isHigh
                        setup.isLow := pivot.isLow
                        setup.active := false
                        setup.pivot := pivot

                        for j = 0 to MS.ST.size() - 1 by 1
                            Pivot p = MS.ST.get(j)
                            if (pivot.isHigh and p.isHigh or pivot.isLow and p.isLow) and p.claimed and not p.reclaimed
                                p.reclaimed := true
                                p.reclaimed

    MS
