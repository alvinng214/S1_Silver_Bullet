// ─── SESSION HELPERS ──────────────────────────────────────────────────────────
getTZ() =>
    "UTC" + (utcOffset >= 0 ? "+" + str.tostring(utcOffset) : str.tostring(utcOffset))

inAsia       = useAsia    and not na(time(timeframe.period, asiaSession,       getTZ()))
inLondon     = useLondon  and not na(time(timeframe.period, londonSession,     getTZ()))
inNewYork    = useNewYork and not na(time(timeframe.period, nySession,         getTZ()))
inCustomExcl = useCustom  and not na(time(timeframe.period, customExclSession, getTZ()))

inAnySession   = inAsia or inLondon or inNewYork
sessionAllowed = not inCustomExcl and (not useSessionFilter or inAnySession)

sessionTag(bool inA, bool inL, bool inNY) =>
    inNY and inL ? "LN+NY" : inL ? "London" : inNY ? "NY" : inA ? "Asia" : "Off-Session"

curSession = sessionTag(inAsia, inLondon, inNewYork)

// ─── EMA ──────────────────────────────────────────────────────────────────────

emaDynColor   = priceAboveEMA ? color.new(#00bcd4, 10) : color.new(#dd326b, 10)
plot(showEma ? ema : na, "EMA Filter", color=emaDynColor, linewidth=1)

// ─── ATR BANDS ────────────────────────────────────────────────────────────────
smoothATR  = ta.ema(ta.tr, 750)

lowerBand  = ema - atrMult * smoothATR

upperBandPlot     = plot(showATRBandLines ? upperBand : na, "Upper ATR Band",            color=upperBandCol,               linewidth=1)
upperBandFillPlot = plot(enableATRFill    ? upperBand : na, "Upper Band (fill ref)",     color=color.new(color.white,100), linewidth=1)

lowerBandPlot     = plot(showATRBandLines ? lowerBand : na, "Lower ATR Band",            color=lowerBandCol,               linewidth=1)
fill(emaRefPlot, upperBandFillPlot, color=enableATRFill ? upperFillCol : color.new(color.white,100), title="Upper ATR Fill")
fill(emaRefPlot, lowerBandFillPlot, color=enableATRFill ? lowerFillCol : color.new(color.white,100), title="Lower ATR Fill")

// ─── SWING DETECTION ──────────────────────────────────────────────────────────

swingLow  = ta.pivotlow (low,  swingLen, swingLen)

var float lastSwingLow     = na
var int   lastSwingHighBar = na
var int   lastSwingLowBar  = na

if not na(swingHigh)
    lastSwingHigh    := swingHigh
    lastSwingHighBar := bar_index - swingLen
if not na(swingLow)
    lastSwingLow    := swingLow
    lastSwingLowBar := bar_index - swingLen

swingHighOB = ta.pivothigh(high, swingLenOB, swingLenOB)
swingLowOB  = ta.pivotlow (low,  swingLenOB, swingLenOB)

var float lastSwingHighOB    = na
var float lastSwingLowOB     = na
var int   lastSwingHighBarOB = na
var int   lastSwingLowBarOB  = na

if not na(swingHighOB)
    lastSwingHighOB    := swingHighOB
    lastSwingHighBarOB := bar_index - swingLenOB
if not na(swingLowOB)
    lastSwingLowOB    := swingLowOB

htfInvalidate(box[] bullBoxes, box[] bearBoxes, label[] bullLabels, label[] bearLabels) =>
    if array.size(bullBoxes) > 0
        for i = array.size(bullBoxes) - 1 to 0
            b      = array.get(bullBoxes, i)
            bLeft  = box.get_left(b)
            bRight = bar_index + 500
            bBot   = box.get_bottom(b)
            midX   = int(bLeft + (bRight - bLeft) / 2)
            lbl    = array.get(bullLabels, i)
            label.set_x(lbl, midX)
            if math.min(open, close) < bBot
                box.set_right(b, bar_index)
                label.set_x(lbl, int(box.get_left(b) + (bar_index - box.get_left(b)) / 2))
                array.remove(bullBoxes,  i)
                array.remove(bullLabels, i)
                box.set_right(b, bar_index)
            else
                box.set_right(b, bRight)
    if array.size(bearBoxes) > 0
        for i = array.size(bearBoxes) - 1 to 0
            b      = array.get(bearBoxes, i)
            bLeft  = box.get_left(b)
            bRight = bar_index + 500
            bTop   = box.get_top(b)
            midX   = int(bLeft + (bRight - bLeft) / 2)
            lbl    = array.get(bearLabels, i)
            label.set_x(lbl, midX)
            if math.max(open, close) > bTop
                box.set_right(b, bar_index)
                label.set_x(lbl, int(box.get_left(b) + (bar_index - box.get_left(b)) / 2))
                array.remove(bearBoxes,  i)
                array.remove(bearLabels, i)
                box.set_right(b, bar_index)
            else
                box.set_right(b, bRight)

// ─── OPEN TRADE FUNCTION ──────────────────────────────────────────────────────
openTrade(array<RRTrade> trades, bool isBull, float slDist, float sl, float tp, float entry, string sess, bool isOBsignal) =>
    slB_bull = box.new(bar_index, entry, bar_index + 1, sl,    bgcolor=color.new(#ff1744, 85), border_color=color.new(#ff1744, 100), border_width=0)
    slB_bear = box.new(bar_index, sl,    bar_index + 1, entry, bgcolor=color.new(#ff1744, 85), border_color=color.new(#ff1744, 100), border_width=0)
    tpB_bull = box.new(bar_index, tp,    bar_index + 1, entry, bgcolor=color.new(#00e676, 85), border_color=color.new(#00e676, 100), border_width=0)
    tpB_bear = box.new(bar_index, entry, bar_index + 1, tp,    bgcolor=color.new(#00e676, 85), border_color=color.new(#00e676, 100), border_width=0)
    slB  = isBull ? slB_bull : slB_bear
    tpB  = isBull ? tpB_bull : tpB_bear
    box.delete(isBull ? slB_bear : slB_bull)
    box.delete(isBull ? tpB_bear : tpB_bull)
    tpPips = math.abs(tp - entry) / pipSize
    slPips = slDist / pipSize
    tpTxt  = showPipLabels ? "+" + str.tostring(tpPips, "#.#") + " pips" : ""
    slTxt  = showPipLabels ? "-" + str.tostring(slPips, "#.#") + " pips" : ""
    tpLbl_bull = label.new(bar_index + 1, tp, tpTxt, style=label.style_label_left, textcolor=color.white, color=showPipLabels ? color.new(#00e676, 50) : color.new(#00e676, 100), size=size.small)
    tpLbl_bear = label.new(bar_index + 1, tp, tpTxt, style=label.style_label_left, textcolor=color.white, color=showPipLabels ? color.new(#00e676, 50) : color.new(#00e676, 100), size=size.small)
    slLbl_bull = label.new(bar_index + 1, sl, slTxt, style=label.style_label_left, textcolor=color.white, color=showPipLabels ? color.new(#ff1744, 50) : color.new(#ff1744, 100), size=size.small)
    slLbl_bear = label.new(bar_index + 1, sl, slTxt, style=label.style_label_left, textcolor=color.white, color=showPipLabels ? color.new(#ff1744, 50) : color.new(#ff1744, 100), size=size.small)
    tpLbl = isBull ? tpLbl_bull : tpLbl_bear
    slLbl = isBull ? slLbl_bull : slLbl_bear
    label.delete(isBull ? tpLbl_bear : tpLbl_bull)
    label.delete(isBull ? slLbl_bear : slLbl_bull)
    array.push(trades, RRTrade.new(slB, tpB, tpLbl, slLbl, sl, tp, entry, sl, isBull, false, false, sess))
    if enableEntryAlert
        sigName = isOBsignal ? "OB Touch" : "Sweep"
        dir     = isBull ? "🟢 LONG" : "🔴 SHORT"
        alert(dir + " ENTRY — " + sigName + " | Entry: " + str.tostring(entry) +
              " | SL: " + str.tostring(sl) + " | TP: " + str.tostring(tp) +
              " | Session: " + sess, alert.freq_once_per_bar_close)

// ─── OPEN TRADES ON SIGNAL ────────────────────────────────────────────────────
noActiveOBTrade = overlapOBTrades or array.size(activeOBTrades) == 0

    slDist = (bearTouchSpikeHigh - entry) + rrSlPips * pipSize
    openTrade(activeOBTrades, false, slDist, entry + slDist, entry - slDist * rrRatio, entry, curSession, true)
if showSweepRR and bullSweep and noActiveSWTrade and sessionAllowed
    entry  = close
    slDist = (entry - low) + sweepRRSlPips * pipSize
    openTrade(activeSWTrades, true,  slDist, entry - slDist, entry + slDist * sweepRRRatio, entry, curSession, false)
if showSweepRR and bearSweep and noActiveSWTrade and sessionAllowed
    entry  = close
    slDist = (high - entry) + sweepRRSlPips * pipSize
    openTrade(activeSWTrades, false, slDist, entry + slDist, entry - slDist * sweepRRRatio, entry, curSession, false)

// ─── HELPER: formatta cella WR ────────────────────────────────────────────────
wrStr(int wins, int tot) =>
    tot > 0 ? str.tostring(math.round((wins / tot) * 100, 1), "#.#") + "%" : "—"

wrColor(int wins, int tot) =>
    tot == 0 ? color.rgb(150, 150, 150) : (wins / tot) >= 0.5 ? color.rgb(0, 230, 118) : color.rgb(255, 70, 70)

diffStr(float wp, float lp) =>
    d = wp - lp
    d >= 0 ? "+" + str.tostring(d, "#.#") : str.tostring(d, "#.#")

diffColor(float wp, float lp) =>
    (wp - lp) >= 0 ? color.rgb(0, 230, 118) : color.rgb(255, 70, 70)

// ─── BEST SESSION LOGIC ───────────────────────────────────────────────────────
bestObSession() =>
    scores = bestByPips ?
      array.from(
        ob_asia_t > 0 ? ob_asia_wp - ob_asia_lp : -999999.0,
        ob_lon_t  > 0 ? ob_lon_wp  - ob_lon_lp  : -999999.0,
        ob_ny_t   > 0 ? ob_ny_wp   - ob_ny_lp   : -999999.0,
        ob_off_t  > 0 ? ob_off_wp  - ob_off_lp  : -999999.0) :
      array.from(
        ob_asia_t > 0 ? ob_asia_w / ob_asia_t : -1.0,
        ob_lon_t  > 0 ? ob_lon_w  / ob_lon_t  : -1.0,
        ob_ny_t   > 0 ? ob_ny_w   / ob_ny_t   : -1.0,
        ob_off_t  > 0 ? ob_off_w  / ob_off_t  : -1.0)
    best = array.indexof(scores, array.max(scores))
    best == 0 ? "Asia" : best == 1 ? "London" : best == 2 ? "NY" : "Off-Session"

bestSwSession() =>
    scores = bestByPips ?
      array.from(
        sw_asia_t > 0 ? sw_asia_wp - sw_asia_lp : -999999.0,
        sw_lon_t  > 0 ? sw_lon_wp  - sw_lon_lp  : -999999.0,
        sw_ny_t   > 0 ? sw_ny_wp   - sw_ny_lp   : -999999.0,
        sw_off_t  > 0 ? sw_off_wp  - sw_off_lp  : -999999.0) :
      array.from(
        sw_asia_t > 0 ? sw_asia_w / sw_asia_t : -1.0,
        sw_lon_t  > 0 ? sw_lon_w  / sw_lon_t  : -1.0,
        sw_ny_t   > 0 ? sw_ny_w   / sw_ny_t   : -1.0,
        sw_off_t  > 0 ? sw_off_w  / sw_off_t  : -1.0)
    best = array.indexof(scores, array.max(scores))
    best == 0 ? "Asia" : best == 1 ? "London" : best == 2 ? "NY" : "Off-Session"

    wins_arr   = array.from(ob_asia_w, sw_asia_w, ob_lon_w,  sw_lon_w,  ob_ny_w,  sw_ny_w,  ob_off_w,      sw_off_w)
    wp_arr     = array.from(ob_asia_wp,sw_asia_wp,ob_lon_wp, sw_lon_wp, ob_ny_wp, sw_ny_wp, ob_off_wp,     sw_off_wp)
    lp_arr     = array.from(ob_asia_lp,sw_asia_lp,ob_lon_lp, sw_lon_lp, ob_ny_lp, sw_ny_lp, ob_off_lp,    sw_off_lp)
    isBestOB   = array.from(true,      false,      true,      false,     true,     false,    true,          false)

    if showSessionStats
        for r = 0 to 7
            sess    = array.get(sessions,   r)
            sig     = array.get(signals,    r)
            tot     = array.get(trades_arr, r)
            wins    = array.get(wins_arr,   r)
            wp      = array.get(wp_arr,     r)
            lp      = array.get(lp_arr,     r)
            isOBrow = array.get(isBestOB,   r)
            bestS   = isOBrow ? bOB : bSW
            highlightActive = isOBrow ? showRR : showSweepRR
            rowBg   = highlightActive and sess == bestS ? #2a2200 : #1a1a1a
            txtC    = highlightActive and sess == bestS ? color.rgb(255, 215, 0) : color.rgb(200, 200, 200)

            table.cell(statsTable, 0, r + 4, sess,              text_color=txtC,               text_size=size.small, bgcolor=rowBg)
            table.cell(statsTable, 1, r + 4, sig,               text_color=txtC,               text_size=size.small, bgcolor=rowBg)
            table.cell(statsTable, 2, r + 4, str.tostring(tot), text_color=wht,                text_size=size.small, bgcolor=rowBg)
            table.cell(statsTable, 3, r + 4, wrStr(wins, tot),  text_color=wrColor(wins, tot), text_size=size.small, bgcolor=rowBg)
            table.cell(statsTable, 4, r + 4, tot > 0 ? str.tostring(wp, "#.#") : "—", text_color=color.rgb(0,230,118),  text_size=size.small, bgcolor=rowBg)
            table.cell(statsTable, 5, r + 4, tot > 0 ? str.tostring(lp, "#.#") : "—", text_color=color.rgb(255,70,70),  text_size=size.small, bgcolor=rowBg)
            table.cell(statsTable, 6, r + 4, tot > 0 ? diffStr(wp, lp) : "—",         text_color=tot > 0 ? diffColor(wp, lp) : grey, text_size=size.small, bgcolor=rowBg)

        bestTxt = (showRR ? "⭐ Best OB Session: " + bOB : "") +
                  (showRR and showSweepRR ? "   |   " : "") +
                  (showSweepRR ? "⭐ Best Sweep Session: " + bSW : "")
        table.cell(statsTable, 0, 12, bestTxt, text_color=gold, text_size=size.small, bgcolor=#1a1a00)
        table.merge_cells(statsTable, 0, 12, 6, 12)

    totRow = showSessionStats ? 13 : 4

    table.cell(statsTable, 0, totRow,     "TOTAL",  text_color=gold, text_size=size.small, bgcolor=dark)
    table.cell(statsTable, 1, totRow,     "🔼 OB",  text_color=color.new(#00bcd4, 0), text_size=size.small, bgcolor=dark)
    table.cell(statsTable, 2, totRow,     str.tostring(obTrades), text_color=wht, text_size=size.small, bgcolor=dark)
    table.cell(statsTable, 3, totRow,     str.tostring(obWR, "#.##") + "%", text_color=obWR >= 50 ? color.rgb(0,230,118) : color.rgb(255,70,70), text_size=size.small, bgcolor=dark)
    table.cell(statsTable, 4, totRow,     str.tostring(obWinPips,  "#.#"), text_color=color.rgb(0,230,118),  text_size=size.small, bgcolor=dark)
    table.cell(statsTable, 5, totRow,     str.tostring(obLossPips, "#.#"), text_color=color.rgb(255,70,70),  text_size=size.small, bgcolor=dark)
    table.cell(statsTable, 6, totRow,     str.tostring(obDiff, "#.#"), text_color=obDiff >= 0 ? color.rgb(0,230,118) : color.rgb(255,70,70), text_size=size.small, bgcolor=dark)

    table.cell(statsTable, 0, totRow + 1, "TOTAL",    text_color=gold, text_size=size.small, bgcolor=mid)
    table.cell(statsTable, 1, totRow + 1, "❌ Sweep", text_color=color.new(#ff9800, 0), text_size=size.small, bgcolor=mid)
    table.cell(statsTable, 2, totRow + 1, str.tostring(swTrades), text_color=wht, text_size=size.small, bgcolor=mid)
    table.cell(statsTable, 3, totRow + 1, str.tostring(swWR, "#.##") + "%", text_color=swWR >= 50 ? color.rgb(0,230,118) : color.rgb(255,70,70), text_size=size.small, bgcolor=mid)
    table.cell(statsTable, 4, totRow + 1, str.tostring(swWinPips,  "#.#"), text_color=color.rgb(0,230,118),  text_size=size.small, bgcolor=mid)
    table.cell(statsTable, 5, totRow + 1, str.tostring(swLossPips, "#.#"), text_color=color.rgb(255,70,70),  text_size=size.small, bgcolor=mid)
    table.cell(statsTable, 6, totRow + 1, str.tostring(swDiff, "#.#"), text_color=swDiff >= 0 ? color.rgb(0,230,118) : color.rgb(255,70,70), text_size=size.small, bgcolor=mid)
