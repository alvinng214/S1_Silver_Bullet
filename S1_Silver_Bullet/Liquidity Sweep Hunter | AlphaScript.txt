//@version=6
indicator("Liquidity Sweep Hunter | AlphaScript", shorttitle="Liquidity Hunter | AlphaScript", overlay=true, max_lines_count=200, max_labels_count=100)

// ============================================================
// INPUTS
// ============================================================
grpSwing   = "Swing Detection"
grpFilter  = "Filters"
grpDisplay = "Display"
grpAlerts  = "Alerts"

swingLen    = input.int(10,    title="Swing Lookback",        minval=3,  maxval=50, group=grpSwing)
sweepMode   = input.string("Both", title="Sweep Mode",       options=["Wick", "Retest", "Both"], group=grpSwing)
maxLevels   = input.int(10,    title="Max Active Levels",     minval=5,  maxval=100, group=grpSwing)
maxAge      = input.int(300,   title="Max Level Age (bars)",  minval=50, maxval=2000, group=grpSwing)
maxSweeps   = input.int(10,    title="Max Visible Sweeps",    minval=1,  maxval=50, group=grpSwing)
recentBars  = input.int(500,   title="Show Signals (recent bars)", minval=50, maxval=5000, group=grpSwing, tooltip="Plotshape triangles only appear on the most recent N bars to reduce chart noise.")

atrLen      = input.int(14,    title="ATR Length",            minval=5,  group=grpFilter)
atrMult     = input.float(0.1, title="Min Wick ATR Mult",     minval=0.0, maxval=2.0, step=0.05, group=grpFilter, tooltip="Minimum wick size beyond level as multiple of ATR. 0 = no filter.")

showLines   = input.bool(true,  title="Show Level Lines",     group=grpDisplay)
showLabels  = input.bool(true,  title="Show Sweep Labels",    group=grpDisplay)
bullCol     = input.color(#00bcd4, title="Bullish Sweep Color", group=grpDisplay)
bearCol     = input.color(#f23645, title="Bearish Sweep Color", group=grpDisplay)
levelCol    = input.color(#787b86, title="Level Line Color",    group=grpDisplay)

// ============================================================
// HELPERS
// ============================================================
atrVal = ta.atr(atrLen)

inRecentWindow = bar_index >= last_bar_index - recentBars

// ============================================================
// STATE — UDT for swing levels
// ============================================================
type SwingLevel
    float  price
    int    lvlTime
    int    lvlBar
    bool   isBull
    bool   broken
    line   lvlLine

var SwingLevel[] bullLevels = array.new<SwingLevel>()
var SwingLevel[] bearLevels = array.new<SwingLevel>()

var line[]  sweepLines  = array.new<line>()
var label[] sweepLabels = array.new<label>()

var line[] bullLevelLines = array.new<line>()
var line[] bearLevelLines = array.new<line>()

// ============================================================
// PIVOT DETECTION (confirmed bars only — repaint-safe)
// ============================================================
ph = ta.pivothigh(high, swingLen, swingLen)
pl = ta.pivotlow(low,  swingLen, swingLen)

if not na(ph) and barstate.isconfirmed
    if showLines and bearLevelLines.size() >= maxLevels
        old = bearLevelLines.shift()
        line.delete(old)
    newLine = showLines ? line.new(
         x1    = time[swingLen],
         y1    = ph,
         x2    = time,
         y2    = ph,
         xloc  = xloc.bar_time,
         extend = extend.right,
         color  = color.new(levelCol, 40),
         style  = line.style_dotted,
         width  = 1) : line(na)
    if showLines
        bearLevelLines.push(newLine)
    lvl = SwingLevel.new(
         price   = ph,
         lvlTime = time[swingLen],
         lvlBar  = bar_index - swingLen,
         isBull  = false,
         broken  = false,
         lvlLine = newLine
    )
    bearLevels.push(lvl)
    if bearLevels.size() > maxLevels
        old = bearLevels.shift()
        line.delete(old.lvlLine)

if not na(pl) and barstate.isconfirmed
    if showLines and bullLevelLines.size() >= maxLevels
        old = bullLevelLines.shift()
        line.delete(old)
    newLine = showLines ? line.new(
         x1    = time[swingLen],
         y1    = pl,
         x2    = time,
         y2    = pl,
         xloc  = xloc.bar_time,
         extend = extend.right,
         color  = color.new(levelCol, 40),
         style  = line.style_dotted,
         width  = 1) : line(na)
    if showLines
        bullLevelLines.push(newLine)
    lvl = SwingLevel.new(
         price   = pl,
         lvlTime = time[swingLen],
         lvlBar  = bar_index - swingLen,
         isBull  = true,
         broken  = false,
         lvlLine = newLine
    )
    bullLevels.push(lvl)
    if bullLevels.size() > maxLevels
        old = bullLevels.shift()
        line.delete(old.lvlLine)

// ============================================================
// SWEEP DETECTION — runs on confirmed bars only
// ============================================================
bullSweep = false
bearSweep = false
bullSweepPrice = 0.0
bearSweepPrice = 0.0

f_capSweepDrawings() =>
    if sweepLabels.size() > maxSweeps
        old = sweepLabels.shift()
        label.delete(old)
    if sweepLines.size() > maxSweeps
        old = sweepLines.shift()
        line.delete(old)

if barstate.isconfirmed
    // --- BEARISH SWEEP: price hunts a swing HIGH (Buy Side Liquidity taken) ---
    if bearLevels.size() > 0
        for i = bearLevels.size() - 1 to 0
            if i < bearLevels.size()
                lvl = bearLevels.get(i)
                age = bar_index - lvl.lvlBar
                if age > maxAge
                    line.delete(lvl.lvlLine)
                    for j = bearLevelLines.size() - 1 to 0
                        if j < bearLevelLines.size()
                            if bearLevelLines.get(j) == lvl.lvlLine
                                _ = bearLevelLines.remove(j)
                    _ = bearLevels.remove(i)
                else
                    wickBeyond = high - lvl.price
                    wickOk     = atrMult == 0.0 or wickBeyond >= atrVal * atrMult
                    swept      = false
                    if (sweepMode == "Wick" or sweepMode == "Both")
                        if high > lvl.price and close < lvl.price and wickOk
                            swept := true
                    if (sweepMode == "Retest" or sweepMode == "Both")
                        if not lvl.broken and close > lvl.price
                            lvl.broken := true
                        else if lvl.broken and close < lvl.price
                            swept := true
                    if swept
                        bearSweep      := true
                        bearSweepPrice := lvl.price
                        line.set_extend(lvl.lvlLine, extend.none)
                        line.set_x2(lvl.lvlLine, time)
                        line.set_color(lvl.lvlLine, color.new(bearCol, 20))
                        line.set_style(lvl.lvlLine, line.style_dashed)
                        line.set_width(lvl.lvlLine, 2)
                        for j = bearLevelLines.size() - 1 to 0
                            if j < bearLevelLines.size()
                                if bearLevelLines.get(j) == lvl.lvlLine
                                    _ = bearLevelLines.remove(j)
                        sweepLines.push(lvl.lvlLine)
                        _ = bearLevels.remove(i)

    // --- BULLISH SWEEP: price hunts a swing LOW (Sell Side Liquidity taken) ---
    if bullLevels.size() > 0
        for i = bullLevels.size() - 1 to 0
            if i < bullLevels.size()
                lvl = bullLevels.get(i)
                age = bar_index - lvl.lvlBar
                if age > maxAge
                    line.delete(lvl.lvlLine)
                    for j = bullLevelLines.size() - 1 to 0
                        if j < bullLevelLines.size()
                            if bullLevelLines.get(j) == lvl.lvlLine
                                _ = bullLevelLines.remove(j)
                    _ = bullLevels.remove(i)
                else
                    wickBeyond = lvl.price - low
                    wickOk     = atrMult == 0.0 or wickBeyond >= atrVal * atrMult
                    swept      = false
                    if (sweepMode == "Wick" or sweepMode == "Both")
                        if low < lvl.price and close > lvl.price and wickOk
                            swept := true
                    if (sweepMode == "Retest" or sweepMode == "Both")
                        if not lvl.broken and close < lvl.price
                            lvl.broken := true
                        else if lvl.broken and close > lvl.price
                            swept := true
                    if swept
                        bullSweep      := true
                        bullSweepPrice := lvl.price
                        line.set_extend(lvl.lvlLine, extend.none)
                        line.set_x2(lvl.lvlLine, time)
                        line.set_color(lvl.lvlLine, color.new(bullCol, 20))
                        line.set_style(lvl.lvlLine, line.style_dashed)
                        line.set_width(lvl.lvlLine, 2)
                        for j = bullLevelLines.size() - 1 to 0
                            if j < bullLevelLines.size()
                                if bullLevelLines.get(j) == lvl.lvlLine
                                    _ = bullLevelLines.remove(j)
                        sweepLines.push(lvl.lvlLine)
                        _ = bullLevels.remove(i)

// ============================================================
// LABELS — BSL for Buy Side Liquidity taken, SSL for Sell Side
// ============================================================
// [MODIFIED] bearSweep = swing HIGH taken = Buy Side Liquidity (BSL)
if bearSweep and showLabels
    lbl = label.new(
         x         = bar_index,
         y         = high,
         text      = "BSL",
         xloc      = xloc.bar_index,
         yloc      = yloc.abovebar,
         color     = color.new(bearCol, 10),
         textcolor = color.white,
         style     = label.style_label_down,
         size      = size.small)
    sweepLabels.push(lbl)
    f_capSweepDrawings()

// [MODIFIED] bullSweep = swing LOW taken = Sell Side Liquidity (SSL)
if bullSweep and showLabels
    lbl = label.new(
         x         = bar_index,
         y         = low,
         text      = "SSL",
         xloc      = xloc.bar_index,
         yloc      = yloc.belowbar,
         color     = color.new(bullCol, 10),
         textcolor = color.white,
         style     = label.style_label_up,
         size      = size.small)
    sweepLabels.push(lbl)
    f_capSweepDrawings()

// ============================================================
// PLOTSHAPE — repaint-safe signals, recent bars only
// ============================================================
plotshape(bullSweep and inRecentWindow, title="SSL — Sell Side Liquidity Taken",
     style    = shape.triangleup,
     location = location.belowbar,
     color    = bullCol,
     size     = size.small,
     text     = "")

plotshape(bearSweep and inRecentWindow, title="BSL — Buy Side Liquidity Taken",
     style    = shape.triangledown,
     location = location.abovebar,
     color    = bearCol,
     size     = size.small,
     text     = "")

// ============================================================
// DASHBOARD TABLE
// ============================================================
var table dash = table.new(position.top_right, 2, 4,
     bgcolor      = color.new(#1e222d, 5),
     border_color = color.new(#363a45, 0),
     border_width = 1)

if barstate.islast
    
    table.cell(dash, 1, 0, syminfo.ticker,   text_color=#ffeb3b, text_size=size.small, bgcolor=color.new(#2a2e39, 0))
    table.cell(dash, 0, 1, "Bull Levels",   text_color=#ffffff, text_size=size.tiny)
    table.cell(dash, 1, 1, str.tostring(bullLevels.size()), text_color=bullCol, text_size=size.tiny)
    table.cell(dash, 0, 2, "Bear Levels",   text_color=#ffffff, text_size=size.tiny)
    table.cell(dash, 1, 2, str.tostring(bearLevels.size()), text_color=bearCol, text_size=size.tiny)
    table.cell(dash, 0, 3, "Mode",          text_color=#ffffff, text_size=size.tiny)
    table.cell(dash, 1, 3, sweepMode,        text_color=#d1d4dc, text_size=size.tiny)

// ============================================================
// ALERTS
// ============================================================
alertcondition(bullSweep, title="SSL — Sell Side Liquidity Taken", message="Sell Side Liquidity swept — swing low taken, price reversed up")
alertcondition(bearSweep, title="BSL — Buy Side Liquidity Taken", message="Buy Side Liquidity swept — swing high taken, price reversed down")
alertcondition(bullSweep or bearSweep, title="Any Liquidity Sweep", message="Liquidity sweep detected")
